age_q2 = df["Age"].quantile(0.50)
age_q3 = df["Age"].quantile(0.75)
age_max = df["Age"].max()

#Years of practice

years_min = df["Years_of_practice"].min()

years_q2 = df["Years_of_practice"].quantile(0.50)
years_q3 = df["Years_of_practice"].quantile(0.75)
years_max = df["Years_of_practice"].max()

print("Years_of_practice - Min:", years_min, "Q1", years_q1, "Q2", years_q2, "Q3", years_q3, "Max", years_max)
